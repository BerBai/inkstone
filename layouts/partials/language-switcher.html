{{/* Language switcher — emits one link per non-current language. Behavior: - If the current page has a translation in , link to that translation - Otherwise, link to the home page of (so readers always land somewhere that matches the new language; never a 404) Bails to nothing when only one language is configured. Default language detection: first entry of site.Sites (sorted by language weight). `label` is read from each language's params, keeping the same template compatible with Hugo versions that reject custom language keys. See partials/head.html hreflang block for the same convention. */}} {{- $cur := . -}} {{- $sites := false -}} {{- if lt hugo.Version "0.156.0" -}}{{- $sites = site.Sites -}}{{- else -}}{{- $sites = hugo.Sites -}}{{- end -}} {{- if gt (len $sites) 1 -}} {{- /* Per-language icon. Keyed by full Hugo Lang code so future regional variants (e.g. en-gb) can map distinctly. Falls back to the generic `languages` glyph when an unmapped language is added — fail visible rather than fail blank. */ -}} {{- $iconMap := dict "zh-cn" "lang-zh" "en" "lang-en" -}} {{- /* translationKey-paired relative URLs for the current page, indexed by Lang. RelPermalink (not Permalink) keeps the link decoupled from baseURL — a placeholder baseURL like https://inkstone.example/ would otherwise punt the user off-site on every language switch. hreflang in head.html still uses Permalink, which is correct for that spec. */ -}} {{- $translations := dict -}} {{- range .Translations -}} {{- $translations = merge $translations (dict .Lang .RelPermalink) -}} {{- end -}}
{{- range $sites -}} {{- $site := . -}} {{- $lang := $site.Language -}} {{- if ne $lang.Lang $cur.Language.Lang -}} {{- $href := index $translations $lang.Lang -}} {{- if not $href -}} {{- /* No translation for this page; link to that language's home, respecting baseURL and language subdirectory config. */ -}} {{- $href = $site.Home.RelPermalink -}} {{- end -}} {{- $iconName := index $iconMap $lang.Lang -}} {{- if not $iconName -}}{{- $iconName = "languages" -}}{{- end -}} {{- $lbl := $lang.Params.label -}} {{- if not $lbl -}}{{- if lt hugo.Version "0.158.0" -}}{{- $lbl = $lang.LanguageName -}}{{- else -}}{{- $lbl = $lang.Label -}}{{- end -}}{{- end -}} {{- if not $lbl -}}{{- $lbl = $lang.Lang -}}{{- end -}} {{ partial "icon.html" $iconName }} {{- end -}} {{- end -}}
{{- end -}}
